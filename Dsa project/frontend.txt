let fetchedLocations = {};

function highlightPath(path, distance) {
  document.querySelectorAll("[id^='node-']").forEach((node) => {
    node.style.fill = "#2563eb";
  });

  document.querySelectorAll("[id^='edge-']").forEach((edge) => {
    edge.style.stroke = "#9499a4";
    edge.style.strokeWidth = "3";
  });

  path.forEach((v) => {
    const node = document.getElementById("node-" + names[v]);
    if (node) node.style.fill = "#69f957";
  });

  for (let i = 0; i < path.length - 1; i++) {
    let u = path[i];
    let v = path[i + 1];

    let edgeId = `edge-${Math.min(u, v)}-${Math.max(u, v)}`;
    const edge = document.getElementById(edgeId);

    if (edge) {
      edge.style.stroke = "#242629";
      edge.style.strokeWidth = "5";
    }
  }

  document.getElementById("distanceMonitor").innerHTML = distance + " m";
}

async function loadLocations() {
  const source = document.getElementById("source");
  const destination = document.getElementById("destination");

  try {
    const response = await fetch(`${api}/locations`);
    if (!response.ok) throw new Error("Failed");

    const data = await response.json();
    source.replaceChildren();
    destination.replaceChildren();

    const sDef = document.createElement("option");
    sDef.value = "-1";
    sDef.textContent = "Source";
    source.appendChild(sDef);

    const dDef = document.createElement("option");
    dDef.value = "-1";
    dDef.textContent = "Destination";
    destination.appendChild(dDef);

    data.locations.forEach((loc) => {
      fetchedLocations[loc.id] = loc.name;

      const opt1 = document.createElement("option");
      opt1.value = loc.id;
      opt1.textContent = loc.name;
      source.appendChild(opt1);

      const opt2 = document.createElement("option");
      opt2.value = loc.id;
      opt2.textContent = loc.name;
      destination.appendChild(opt2);
    });
  } catch (error) {
    console.error("Error loading locations:", error);
  }
}

const findPath = async () => {
  const source = Number(document.getElementById("source").value);
  const destination = Number(document.getElementById("destination").value);

  if (source >= 0 && destination >= 0) {
    try {
      const response = await fetch(`${api}/find-path`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ source, destination }),
      });

      if (!response.ok) throw new Error("Backend request failed");

      const data = await response.json();
      highlightPath(data.path, data.distance);
    } catch (error) {
      console.error(error);
    }
  } else {
    alert("Please enter source & destination");
  }
};

const search = () => {
  const input = document.getElementById("searchInput").value.trim();
  const searchList = document.getElementById("searchList");
  searchList.innerHTML = "";

  const entries = Object.values(fetchedLocations);
  const filtered = input === ""
    ? entries
    : entries.filter((item) => item.toLowerCase().includes(input.toLowerCase()));

  filtered.forEach((item) => {
    const li = document.createElement("li");
    li.textContent = item;
    searchList.appendChild(li);
  });
};

window.addEventListener("load", async () => {
  await loadLocations();
  search();
  document.getElementById("searchInput").addEventListener("input", search);
});

const svg = document.getElementById("campusMap");
const mapContent = document.getElementById("mapContent");

const zoomInBtn = document.getElementById("zoomIn");
const zoomOutBtn = document.getElementById("zoomOut");
const resetBtn = document.getElementById("resetMap");

let scale = 1;
let x = 0;
let y = 0;

const MIN_ZOOM = 0.5;
const MAX_ZOOM = 4;
const ZOOM_STEP = 1.2;

function updateMap() {
  mapContent.setAttribute("transform", `translate(${x} ${y}) scale(${scale})`);
}

function getSvgPoint(event) {
  const point = svg.createSVGPoint();

  point.x = event.clientX;
  point.y = event.clientY;

  return point.matrixTransform(svg.getScreenCTM().inverse());
}

// নির্দিষ্ট SVG position-কে কেন্দ্র করে zoom
function zoomAt(newScale, px, py) {
  newScale = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, newScale));

  // Zoom-এর সময় pointer-এর নিচের জায়গা স্থির রাখে
  x = px - (px - x) * (newScale / scale);
  y = py - (py - y) * (newScale / scale);

  scale = newScale;
  updateMap();
}

// Zoom In
zoomInBtn.addEventListener("click", () => {
  zoomAt(scale * ZOOM_STEP, 400, 250);
});

// Zoom Out
zoomOutBtn.addEventListener("click", () => {
  zoomAt(scale / ZOOM_STEP, 400, 250);
});

// Reset
resetBtn.addEventListener("click", () => {
  scale = 1;
  x = 0;
  y = 0;
  updateMap();
});

// Mouse wheel zoom
svg.addEventListener(
  "wheel",
  (event) => {
    event.preventDefault();

    const point = getSvgPoint(event);

    const factor = event.deltaY < 0 ? ZOOM_STEP : 1 / ZOOM_STEP;

    zoomAt(scale * factor, point.x, point.y);
  },
  { passive: false },
);

// Drag / Pan
let dragging = false;
let lastPoint = null;

svg.addEventListener("pointerdown", (event) => {
  // Button বা অন্য interactive element থেকে drag শুরু নয়
  if (event.target.closest("button, a")) return;

  dragging = true;
  lastPoint = getSvgPoint(event);

  svg.setPointerCapture(event.pointerId);
  svg.classList.add("dragging");
});

svg.addEventListener("pointermove", (event) => {
  if (!dragging) return;

  const point = getSvgPoint(event);

  x += point.x - lastPoint.x;
  y += point.y - lastPoint.y;

  lastPoint = point;
  updateMap();
});

function stopDragging() {
  dragging = false;
  lastPoint = null;
  svg.classList.remove("dragging");
}

svg.addEventListener("pointerup", stopDragging);
svg.addEventListener("pointercancel", stopDragging);

// প্রথম অবস্থায় map দেখাও
updateMap();

void addVertex(int v1, int v2, int weight, vector<vector<pair<int, int>>> &graph)
{
    graph[v1].push_back({v2, weight});
    graph[v2].push_back({v1, weight});
}

pair<vector<int>, int> dijkstra(int source, int destination, vector<vector<pair<int, int>>> &graph)
{
    int n = graph.size();
    vector<int> dist(n, INF);
    vector<int> parent(n, -1);

    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;

    dist[source] = 0;
    pq.push({0, source});

    while (!pq.empty())
    {
        int currentDist = pq.top().first;
        int u = pq.top().second;
        pq.pop();

        if (currentDist != dist[u])
            continue;

        if (u == destination)
            break;

        for (auto edge : graph[u])
        {
            int v = edge.first;
            int weight = edge.second;

            if (dist[u] + weight < dist[v])
            {
                dist[v] = dist[u] + weight;
                parent[v] = u;
                pq.push({dist[v], v});
            }
        }
    }

    if (dist[destination] == INF)
    {
        return {{}, 0};
    }

    vector<int> path;
    for (int v = destination; v != -1; v = parent[v])
    {
        path.push_back(v);
    }

    vector<int> exactPath;
    for (int i = path.size() - 1; i >= 0; i--)
    {
        exactPath.push_back(path[i]);
    }

    return {exactPath, dist[destination]};
}
