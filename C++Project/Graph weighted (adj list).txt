#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int vertex = 4;
    vector<pair<int, int>> graph[vertex];

    // Edge connection
    graph[0].push_back({1, 1});
    graph[0].push_back({2, 1});
    graph[0].push_back({3, 3});

    graph[1].push_back({0, 1});
    graph[1].push_back({2, 2});

    graph[2].push_back({0, 1});
    graph[2].push_back({1, 2});
    graph[2].push_back({3, 2});

    graph[3].push_back({0, 3});
    graph[3].push_back({2, 2});

    // cout << "connection = " << graph[3][0].first << endl;
    // cout << "cost = " << graph[3][0].second << endl;

    for (int i = 0; i < vertex; i++)
    {
        cout << i << "-> ";
        for (int j = 0; j < graph[i].size(); j++)
        {
            int con = graph[i][j].first;
            int cost = graph[i][j].second;
            cout << "\n[Connection: " << con << ",Cost: " << cost << "] ";
        }
        cout << endl;
    }

    return 0;
}

#include <iostream>
#include <vector>
using namespace std;

void addVertext(int v1, int v2, int weight, vector<vector<pair<int, int>>> &graph)
{
    graph[v1].push_back({v2, weight});
    // graph[v2].push_back({v1, weight});
}

int main()
{
    int vertex = 4;
    vector<vector<pair<int, int>>> graph(vertex);

    // Vertex connection
    addVertext(0, 1, 4, graph);
    addVertext(0, 2, 2, graph);

    addVertext(1, 0, 4, graph);
    addVertext(1, 2, 1, graph);
    addVertext(1, 3, 5, graph);

    addVertext(2, 0, 2, graph);
    addVertext(2, 1, 1, graph);
    addVertext(2, 3, 8, graph);

    addVertext(3, 1, 5, graph);
    addVertext(3, 2, 8, graph);

    for (int i = 0; i < vertex; i++)
    {
        for (int j = 0; j < graph[i].size(); j++)
        {
            int con = graph[i][j].first, weight = graph[i][j].second;
            cout << i << " -> " << con << " = " << weight << " km" << endl;
        }
    }

    return 0;
}
