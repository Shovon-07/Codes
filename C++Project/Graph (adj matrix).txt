#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int vertex = 4;
    int graph[4][4] = {0};

    // Edge connection
    graph[0][1] = 1;
    graph[0][2] = 1;

    graph[1][0] = 1;
    graph[1][2] = 1;

    graph[2][0] = 1;
    graph[2][1] = 1;
    graph[2][3] = 1;

    graph[3][2] = 1;

    for (int i = 0; i < vertex; i++)
    {
        for (int j = 0; j < vertex; j++)
        {
            cout << graph[i][j] << " ";
        }
        cout << endl;
    }

    return 0;
}
