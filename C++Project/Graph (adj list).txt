#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int vertex = 4;
    vector<int> graph[vertex];

    // Edge connection
    graph[0].push_back(1);
    graph[0].push_back(2);

    graph[1].push_back(0);
    graph[1].push_back(2);

    graph[2].push_back(0);
    graph[2].push_back(1);
    graph[2].push_back(3);

    graph[3].push_back(2);

    for (int i = 0; i < vertex; i++)
    {
        cout << i << "-> ";
        for (int j = 0; j < graph[i].size(); j++)
        {
            cout << graph[i][j] << " ";
        }
        cout << endl;
    }

    return 0;
}
